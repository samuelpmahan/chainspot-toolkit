const event=(type,role=null,status=null)=>({type,role,status,seq:null,parent:null,depth:null,captureIndex:null,
  errorCode:/(_FAIL|_THROW)$/.test(type)?'E_FIXTURE':null,errorText:/(_FAIL|_THROW)$/.test(type)?'fixture failure':null});
const native=(role,status)=>({...event('NATIVE_EXIT',role,status),seq:role==='constant'?1:0,parent:role==='constant'?0:null,
  depth:role==='constant'?1:0,captureIndex:role==='constant'?0:1,errorCode:status==='threw'?'E_FIXTURE':null,errorText:status==='threw'?'fixture native failure':null});
function hook(role){return [event('HOOK_ENTER',role),event('HOOK_ENCODE_START',role),event('HOOK_ENCODE_OK',role),event('HOOK_RETURN',role)];}
function artifact(role){return [event('WRITE_START',role),event('WRITE_OK',role),event('HASH_START',role),event('HASH_OK',role)];}
function returned(){return [event('INTENT'),event('PUBLIC_ENTER'),native('constant','ok'),...hook('constant'),
  native('query','ok'),...hook('query'),event('PUBLIC_RETURN')];}
function answer(){return [event('ANSWER_VALIDATE_START'),event('ANSWER_VALIDATE_OK'),event('ANSWER_ENCODE_START'),event('ANSWER_ENCODE_OK')];}
function success(){return [...returned(),...answer(),...artifact('constant'),...artifact('query'),...artifact('answer'),event('POSTCHECK_START'),event('POSTCHECK_OK')];}
test('M2 all terminal success layers are necessary and individually retained',()=>{
  const r=eventFacts(success(),{termination:'NORMAL'});assert.equal(r.status,'COMPLETE');
  assert.equal(r.publicCall,'RETURNED');assert.equal(r.artifacts.answer.bound,true);
  assert.equal(r.hooks.constant.native,'ok');assert.equal(r.hooks.constant.encode,'OK');
});
test('M2 nested constant hook failure retains constant native-ok, outer native-threw and public throw',()=>{
  const e=[event('INTENT'),event('PUBLIC_ENTER'),native('constant','ok'),event('HOOK_ENTER','constant'),
    event('HOOK_ENCODE_START','constant'),event('HOOK_ENCODE_FAIL','constant'),event('HOOK_THROW','constant'),
    native('query','threw'),...hook('query'),event('PUBLIC_THROW'),event('POSTCHECK_START'),event('POSTCHECK_OK')];
  const r=eventFacts(e,{termination:'NORMAL'});assert.equal(r.status,'FAILED');assert.equal(r.publicCall,'THREW');
  assert.equal(r.hooks.constant.native,'ok');assert.equal(r.hooks.query.native,'threw');
  assert.equal(r.firstFailure.type,'HOOK_ENCODE_FAIL');assert.equal(r.artifacts.answer.bound,false);
});
test('M2 query hook throws after native-ok without falsely recording public return',()=>{
  const e=returned().slice(0,-2);e.push(event('HOOK_THROW','query'),event('PUBLIC_THROW'),event('POSTCHECK_START'),event('POSTCHECK_OK'));
  const r=eventFacts(e,{termination:'NORMAL'});assert.equal(r.status,'FAILED');assert.equal(r.hooks.query.native,'ok');assert.equal(r.publicCall,'THREW');
});
test('M2 bounded empty failure text is retained without invented fallback wording',()=>{

test('M2 every failing terminal layer has a direct falsifier',()=>{
  for(const [start,fail]of [['ANSWER_VALIDATE_START','ANSWER_VALIDATE_FAIL'],['POSTCHECK_START','POSTCHECK_FAIL']]){
    const r=eventFacts([...returned(),event(start),event(fail)],{termination:'NORMAL'});assert.equal(r.status,'FAILED');assert.equal(r.firstFailure.type,fail);
  }
  const r=eventFacts([...returned(),...answer(),event('WRITE_START','answer'),event('WRITE_FAIL','answer')],{termination:'NORMAL'});
  assert.equal(r.status,'FAILED');assert.equal(r.artifacts.answer.write,'FAIL');

    :[event('WRITE_START',role),event('WRITE_OK',role),event('HASH_START',role),event('HASH_FAIL',role)];

  assert.equal(r.status,'FAILED');assert.equal(r.artifacts[role][stage.toLowerCase()],'FAIL');assert.equal(r.artifacts[role].bound,false);
