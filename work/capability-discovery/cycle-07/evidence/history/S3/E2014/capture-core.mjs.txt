/** Cap/pipe/setup errors stop a fixture or real child; only child close supplies exit facts. */
export async function captureSession(io,{streamBytes=65536,wallMs=120000,graceMs=2000}={}){
  const streams={},laterErrors=[];let firstCause=null,child=null,wall=null,grace=null,wait=null,closed=false,unsubscribe=null;
  let actualChildExit={code:null,signal:null,closeObserved:false};
  const record=(stage,e)=>{const item={stage,error:errorInfo(e)};if(firstCause===null)firstCause=item;else if(laterErrors.length<12)laterErrors.push(item);};
  const stop=(stage,e=Error(stage))=>{record(stage,e);if(!child||closed)return;try{child.kill('SIGTERM');}catch(x){record('TERM_REQUEST',x);}if(grace===null)try{grace=io.setTimer(()=>{try{if(!closed)child.kill('SIGKILL');}catch(x){record('KILL_REQUEST',x);}},graceMs);}catch(x){record('KILL_TIMER',x);}};
  try{
    if(io.onStop)unsubscribe=io.onStop(stage=>stop(stage));
    for(const name of ['stdout','stderr']){const fd=io.open(name);streams[name]={fd,receivedBytes:0,writtenBytes:0,prefixOnly:false,writeFailed:false};}
    child=io.spawn();
    wait=new Promise(done=>child.once('close',(code,signal)=>{closed=true;actualChildExit={code,signal,closeObserved:true};done();}));
    child.on('error',e=>stop('SPAWN_ERROR',e));
    wall=io.setTimer(()=>stop('WALL_TIMEOUT'),wallMs);
    for(const name of ['stdout','stderr']){
      const s=streams[name],pipe=child[name],label=name.toUpperCase();
      pipe.on('error',e=>{s.prefixOnly=true;stop(label+'_ERROR',e);});
      pipe.on('data',b=>{s.receivedBytes+=b.length;if(s.writeFailed)return;const take=Math.min(b.length,streamBytes-s.writtenBytes);try{let n=0;while(n<take){const got=io.write(s.fd,b,n,take-n);if(!Number.isSafeInteger(got)||got<1||got>take-n)fail('capture write count');n+=got;s.writtenBytes+=got;}}catch(e){s.writeFailed=true;s.prefixOnly=true;stop(label+'_WRITE',e);}if(take<b.length){s.prefixOnly=true;stop(label+'_CAP');}});
    }
  }catch(e){stop('SETUP',e);}
  // A post-spawn setup failure must NOT jump straight to timer/descriptor cleanup.
  if(wait)try{await wait;}catch(e){record('CLOSE_OBSERVATION_FAILED',e);}
  if(!child||closed){if(wall!==null)io.clearTimer(wall);if(grace!==null)io.clearTimer(grace);}
  if(unsubscribe)try{unsubscribe();}catch(e){record('SIGNAL_CLEANUP',e);}
  for(const s of Object.values(streams)){try{io.sync(s.fd);}catch(e){record('CAPTURE_SYNC',e);}try{io.close(s.fd);}catch(e){record('CAPTURE_CLOSE',e);}}
  return {actualChildExit,firstCause,laterErrors,streams:Object.fromEntries(Object.entries(streams).map(([k,s])=>[k,{receivedBytes:s.receivedBytes,writtenBytes:s.writtenBytes,prefixOnly:s.prefixOnly,writeFailed:s.writeFailed}])),timersCleared:!child||closed};
}

function holdProjection(e){
  const p=e.integrityHold;fields(p,['code','j','c','b','role','side','stationIndex','planIndex','rawValue','answerBinding','rawAnswer','originalState','originalCallbackStatus']);
  if(p.code!=='CONTRAST_INFINITY'||!Number.isInteger(p.c)||p.c<0||p.c>15||!Number.isInteger(p.b)||p.b<0||p.b>2||p.j!==3*p.c+p.b||p.role!=='control'||!['leftLum','rightLum'].includes(p.side)||p.planIndex!==10*p.c+2+3*p.b||(p.rawValue!==Infinity&&p.rawValue!==-Infinity)||p.originalState!=='COMPLETED'||p.originalCallbackStatus!=='ok')fail('HOLD location/source');
  const m=p.answerBinding;fields(m,['path','bytes','sha256','kind','comparisonId','planIndex']);if(!/^fragments\/answer-\d+\.json$/.test(m.path)||!Number.isInteger(m.bytes)||m.bytes<1||m.bytes>8192||!/^[0-9a-f]{64}$/.test(m.sha256)||m.kind!=='answer'||m.planIndex!==p.planIndex||typeof m.comparisonId!=='string'||m.comparisonId.length>128)fail('HOLD answer binding');

}
/** Internal capability seam. It has no way to discover/import a real helper or open evidence. */
export async function invokeOnce(io,ids){
  const helperCall={attempted:false,returned:false,threw:false},postErrors=[];let status='FAILED',plan=null,error=null,holdRetention={requested:false,complete:false,artifact:null,error:null};
  try{
    await io.claim();const module=await io.loadHelper();if(typeof module.buildOriginalEvidencePlan!=='function')fail('helper export');let r;
    helperCall.attempted=true;try{r=await module.buildOriginalEvidencePlan();helperCall.returned=true;}catch(e){helperCall.threw=true;throw e;}

    await io.write('plan.wire.json',r.encodedBytes);plan={path:'plan.wire.json',bytes:r.encodedBytes.length,sha256:r.sha256};status='READY';
  }catch(e){
    error=errorInfo(e);
    if(e?.code==='INTEGRITY_HOLD'){
      holdRetention.requested=true;
      try{const p=holdProjection(e),codec=await io.loadCodec(),b=codec.encodeBytes(p,{maxBytes:65536,maxNodes:8192,maxDepth:32,maxTypedBytes:1});if(!b||Object.getPrototypeOf(b)!==Uint8Array.prototype||b.length>65536)fail('HOLD artifact cap');await io.write('integrity-hold.wire.json',b);holdRetention.complete=true;holdRetention.artifact={path:'integrity-hold.wire.json',bytes:b.length,sha256:io.hash(b)};}catch(x){holdRetention.error=errorInfo(x);postErrors.push({stage:'HOLD_RETENTION',error:errorInfo(x)});}
    }
  }
  const receipt={schema:'cycle07-original-plan-helper-receipt@2',...ids,status,helperCall,plan,error,holdRetention,postErrors,intendedExitCode:status==='READY'?0:1,qualification:'Source-derived helper evidence only; HOLD numbers live solely in lossless artifact, not this JSON.'};

  return {...receipt,receiptPersisted};
}
