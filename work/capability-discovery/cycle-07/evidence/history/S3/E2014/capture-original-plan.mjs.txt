export const digest=b=>createHash('sha256').update(b).digest('hex');
export function fail(s){throw Error('REFUSED: '+s);}
export function hashText(s){if(typeof s!=='string'||!/^[0-9a-f]{64}$/.test(s)||/^0{64}$|^f{64}$/.test(s))fail('final SHA256');return s;}
export function noLinks(path){const a=resolve(path);let p=sep;const parts=a.split(sep).filter(Boolean);for(let i=0;i<parts.length;i++){p=resolve(p,parts[i]);const s=lstatSync(p);if(s.isSymbolicLink()||(i<parts.length-1&&!s.isDirectory()))fail('symlink/path');}return a;}
function descriptor(path,cap,bytes){const p=noLinks(path),before=lstatSync(p);if(!before.isFile()||!Number.isSafeInteger(before.size)||before.size<0||before.size>cap||(bytes!==null&&before.size!==bytes)||!Number.isSafeInteger(constants.O_NOFOLLOW))fail('regular bounded file');const fd=openSync(p,constants.O_RDONLY|constants.O_NONBLOCK|constants.O_NOFOLLOW);try{const s=fstatSync(fd);if(!s.isFile()||s.dev!==before.dev||s.ino!==before.ino||s.size!==before.size)fail('descriptor identity');return {fd,stat:s};}catch(e){closeSync(fd);throw e;}}
function unchanged(fd,s){const a=fstatSync(fd);if(a.size!==s.size||a.dev!==s.dev||a.ino!==s.ino||a.mtimeMs!==s.mtimeMs)fail('changed file');}
export function readExact(path,cap,bytes=null,expected=null){const {fd,stat}=descriptor(path,cap,bytes);try{const b=new Uint8Array(stat.size);let n=0;while(n<b.length){const got=readSync(fd,b,n,b.length-n,n);if(!got)fail('truncated file');n+=got;}if(readSync(fd,new Uint8Array(1),0,1,b.length)!==0)fail('growing file');unchanged(fd,stat);if(expected!==null&&digest(b)!==expected)fail('file hash');return b;}finally{closeSync(fd);}}
function runtimeHash(p){fields(p,['bytes','sha256']);if(!Number.isSafeInteger(p.bytes)||p.bytes<1||p.bytes>LIMIT.runtimeBinary)fail('runtime byte pin');hashText(p.sha256);const {fd,stat}=descriptor(process.execPath,LIMIT.runtimeBinary,p.bytes);try{const h=createHash('sha256'),b=new Uint8Array(65536);let pos=0;while(pos<stat.size){const n=readSync(fd,b,0,Math.min(b.length,stat.size-pos),pos);if(!n)fail('runtime truncated');h.update(b.subarray(0,n));pos+=n;}if(readSync(fd,b,0,1,pos)!==0)fail('runtime grew');unchanged(fd,stat);if(h.digest('hex')!==p.sha256)fail('runtime hash');}finally{closeSync(fd);}}
export const parse=b=>JSON.parse(new TextDecoder('utf-8',{fatal:true}).decode(b));
export function jsonBytes(v){const b=new TextEncoder().encode(`${JSON.stringify(v,null,2)}\n`);if(b.length>LIMIT.metadata)fail('metadata cap');return b;}
export function putNew(path,b,sync=false){let fd;try{fd=openSync(path,'wx',0o600);let n=0;while(n<b.length){const got=writeSync(fd,b,n,b.length-n);if(!got)fail('zero write');n+=got;}if(sync)fsyncSync(fd);}finally{if(fd!==undefined)closeSync(fd);}}
export const putJSON=(path,v,sync=false)=>putNew(path,jsonBytes(v),sync);
export function syncDir(path){const fd=openSync(noLinks(path),'r');try{fsyncSync(fd);}finally{closeSync(fd);}}
function pin(p,path){fields(p,['bytes','sha256']);if(!Number.isSafeInteger(p.bytes)||p.bytes<1||p.bytes>LIMIT.source)fail('final source pin');hashText(p.sha256);readExact(path,LIMIT.source,p.bytes,p.sha256);}

export function cliSha(args){if(args.length!==2||args[0]!=='--packet-sha')fail('usage: --packet-sha FINAL_ROOT_PACKET_SHA256');return hashText(args[1]);}
function parentDir(path){try{mkdirSync(path,0o700);}catch(e){if(e.code!=='EEXIST')throw e;}if(!lstatSync(noLinks(path)).isDirectory())fail('parent directory');}
