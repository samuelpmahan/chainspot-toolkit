export const digest=b=>createHash('sha256').update(b).digest('hex');
export function fail(message){throw new Error(`REFUSED: ${message}`);}
export function exact(o,keys){if(!o||Object.getPrototypeOf(o)!==Object.prototype)fail('ordinary record');const k=Reflect.ownKeys(o);if(k.length!==keys.length||k.some(x=>typeof x!=='string'||!keys.includes(x)))fail('closed fields');for(const x of keys){const d=Object.getOwnPropertyDescriptor(o,x);if(!d||!Object.hasOwn(d,'value')||!d.enumerable||d.value===undefined)fail('own data fields');}}
export function hashText(s){if(typeof s!=='string'||!/^[0-9a-f]{64}$/.test(s)||/^0{64}$|^f{64}$/.test(s))fail('final SHA256 required');return s;}
export function noLinks(path){const a=resolve(path);let p=sep;const parts=a.split(sep).filter(Boolean);for(let i=0;i<parts.length;i++){p=resolve(p,parts[i]);const s=lstatSync(p);if(s.isSymbolicLink()||(i<parts.length-1&&!s.isDirectory()))fail('symlink/non-directory component');}return a;}
export function readExact(path,cap,bytes=null,expected=null){const a=noLinks(path),before=lstatSync(a);if(!before.isFile()||!Number.isSafeInteger(before.size)||before.size<0||before.size>cap||(bytes!==null&&before.size!==bytes))fail('file size/type');if(!Number.isSafeInteger(constants.O_NOFOLLOW))fail('O_NOFOLLOW required');let fd;try{fd=openSync(a,constants.O_RDONLY|constants.O_NONBLOCK|constants.O_NOFOLLOW);const s=fstatSync(fd);if(!s.isFile()||s.dev!==before.dev||s.ino!==before.ino||s.size!==before.size)fail('descriptor identity');const out=new Uint8Array(s.size);let n=0;while(n<out.length){const got=readSync(fd,out,n,out.length-n,n);if(!got)fail('truncated file');n+=got;}if(readSync(fd,new Uint8Array(1),0,1,out.length)!==0)fail('growing file');const after=fstatSync(fd);if(after.size!==s.size||after.mtimeMs!==s.mtimeMs||after.dev!==s.dev||after.ino!==s.ino)fail('file changed');if(expected!==null&&digest(out)!==expected)fail('file hash');return out;}finally{if(fd!==undefined)closeSync(fd);}}
export function parse(bytes){return JSON.parse(new TextDecoder('utf-8',{fatal:true}).decode(bytes));}
export function jsonBytes(value){const b=new TextEncoder().encode(`${JSON.stringify(value,null,2)}\n`);if(b.length>LIMIT.metadata)fail('metadata cap');return b;}
export function putNew(path,bytes,sync=false){let fd;try{fd=openSync(path,'wx',0o600);let n=0;while(n<bytes.length){const got=writeSync(fd,bytes,n,bytes.length-n);if(!got)fail('zero write');n+=got;}if(sync)fsyncSync(fd);}finally{if(fd!==undefined)closeSync(fd);}}
export function putJSON(path,value,sync=false){putNew(path,jsonBytes(value),sync);}
function syncDir(path){const fd=openSync(noLinks(path),'r');try{fsyncSync(fd);}finally{closeSync(fd);}}
export function smallError(e){const full=String(e?.message??e);return {name:String(e?.name??'Error').slice(0,128),message:full.slice(0,4096),messageTruncated:full.length>4096};}
function reviewPin(r,targets){exact(r,['path','bytes','sha256','reviewer','decision','targets']);if(r.decision!=='accepted'||typeof r.reviewer!=='string'||!/^[A-Za-z0-9_-]{1,64}$/.test(r.reviewer)||r.reviewer==='S3')fail('different-worker accepted final review');if(!Array.isArray(r.targets)||JSON.stringify([...r.targets].sort())!==JSON.stringify([...targets].sort()))fail('review targets');if(typeof r.path!=='string'||!r.path.startsWith(`${ROOT}/discovery-cycle07/`)||resolve(r.path)!==r.path)fail('review path');pin(r,r.path);}
function pin(p,path){if(!Number.isSafeInteger(p.bytes)||p.bytes<1||p.bytes>LIMIT.source)fail('final pin bytes');hashText(p.sha256);readExact(path,LIMIT.source,p.bytes,p.sha256);}

function ensureParent(path){try{mkdirSync(path,0o700);}catch(e){if(e.code!=='EEXIST')throw e;}if(!lstatSync(noLinks(path)).isDirectory())fail('fixed parent directory');}

  // Token remains spent after every subsequent failure. No cleanup/overwrite path.
  putJSON(a.tokenPath,{schema:'cycle07-original-plan-spent@1',state:'CONSUMED',runId:a.packet.runId,packetSha256:a.packetSha256,requestSha256:a.requestSha256,pins:a.packet.pins,reviews:a.packet.reviews,wrapperPid:process.pid,consumedUTC:new Date().toISOString()},true);syncDir(`${HERE}/tokens`);

  const stop=reason=>{if(!terminal)terminal=reason;if(child){child.kill('SIGTERM');if(killer===null)killer=setTimeout(()=>child.kill('SIGKILL'),LIMIT.killGraceMs);}};
  for(const name of ['stdout','stderr']){const fd=openSync(`${a.runDir}/${name}.raw`,'wx',0o600);streams[name]={fd,written:0,received:0,prefixOnly:false,writeFailed:false};}

  let observed={code:null,signal:null,closeObserved:false};
  try{
    child=spawn(process.execPath,[a.paths.runner,'--packet-sha',a.packetSha256],{cwd:ROOT,env:{TZ:'UTC'},stdio:['ignore','pipe','pipe'],shell:false});
    const timer=setTimeout(()=>{timedOut=true;stop('WALL_TIMEOUT');},LIMIT.wallMs);

    observed=await new Promise(resolveDone=>{child.on('error',e=>{spawnError=smallError(e);note(e);});child.on('close',(code,signal)=>resolveDone({code,signal,closeObserved:true}));});
    clearTimeout(timer);
  }catch(e){note(e);stop('SPAWN_OR_WAIT_FAILED');}finally{if(killer!==null)clearTimeout(killer);process.off('SIGINT',onINT);process.off('SIGTERM',onTERM);for(const s of Object.values(streams)){try{fsyncSync(s.fd);}catch(e){note(e);}try{closeSync(s.fd);}catch(e){note(e);}}}

  const inventory=[];let receipt=null,planVerified=false,inventoryComplete=false;

  const success=observed.closeObserved&&observed.code===0&&observed.signal===null&&!terminal&&!spawnError&&errors.length===0&&receipt?.status==='READY'&&planVerified;
  const streamMetadata=Object.fromEntries(Object.entries(streams).map(([k,s])=>[k,{receivedBytes:s.received,writtenBytes:s.written,prefixOnly:s.prefixOnly,writeFailed:s.writeFailed}]));
  const exit={schema:'cycle07-original-plan-observed-exit@1',runId:a.packet.runId,packetSha256:a.packetSha256,requestSha256:a.requestSha256,actualChildExit:observed,spawnError,timedOut,interrupted,terminal,errors,streams:streamMetadata,helperReceipt:receipt,planVerified,availableFilesBeforeExit:inventory,inventoryComplete,inventoryExcludes:['exit.json'],wrapperIntendedExitCode:success?0:1,limits:LIMIT,qualification:'Observed child close/exit/signal; wrapper own exit needs external launcher evidence. }
