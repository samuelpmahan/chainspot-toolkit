/** Descriptor-bounded streaming hash. Retain bodies only for explicit small JSON/wires. */
function readBound(path,maxBytes,expected=null,retain=false,exact=null){

  try{fd=openSync(path,constants.O_RDONLY|constants.O_NONBLOCK);const stat=fstatSync(fd),n=stat.size;need(stat.isFile()&&Number.isSafeInteger(n)&&n>=0&&n<=maxBytes&&(exact===null||n===exact),'refused regular-file size');
    const hash=createHash('sha256'),body=retain?new Uint8Array(n):null,chunk=new Uint8Array(Math.max(1,Math.min(n,65536)));let position=0;
    while(position<n){const count=readSync(fd,chunk,0,Math.min(chunk.length,n-position),position);need(count>0,'file shortened during read');hash.update(chunk.subarray(0,count));if(body)body.set(chunk.subarray(0,count),position);position+=count;}

}

export async function renderTrustedPacket({packetPath,packetSha256,outDir}){
  const bundle=verifyPacket(resolve(packetPath),packetSha256);
  // Every externally trusted wire/provenance/artifact digest has passed before this import/decode.
  const {decodeBytes}=await import(CODEC_URL),trace=decodeBytes(bundle.bindings.trace.body),sidecar=decodeBytes(bundle.bindings.sidecar.body,{maxBytes:4194304});
  const model=validate(trace,sidecar),records=displayRecords(model),offset=residuals(model),directory=resolve(outDir);
  const renderer=readBound(fileURLToPath(import.meta.url),131072,null,false),artifacts=[];
  // All content prepared from the same model; destination remains absent until input validation passes.
  const outputs=[['descriptive-table.csv',csv(records)],...GROUPS.map(([g])=>[`group-${g}.svg`,svgGroup(model,g,bundle.bindings)]),['report.html',html(model,records,offset,bundle,directory)]];
  mkdirSync(directory);
  for(const [path,content]of outputs){const bytes=new TextEncoder().encode(content);writeFileSync(join(directory,path),bytes,{flag:'wx'});artifacts.push({path,bytes:bytes.length,sha256:sha(bytes)});}
  const manifest={schema:'cycle06-lenard5-trace-display@1',status:'SVG_CSV_HTML_WRITTEN_NO_PNG_OR_VISUAL_QA',renderer:{path:renderer.path,sha256:renderer.sha256},codec:{path:bundle.codec.path,sha256:CODEC_SHA},bindings:Object.fromEntries(Object.entries(bundle.bindings).map(([k,v])=>[k,{path:v.path,bytes:v.bytes,sha256:v.sha256}])),artifacts,warnings:model.warnings,claims:{promoted:false,heldout:false,pathIdentity:'UNKNOWN',physicalEdgeIdentity:'UNKNOWN',continuity:'UNKNOWN',calibratedWidth:'UNKNOWN',scientificAcceptance:'UNKNOWN',humanAcceptance:'UNKNOWN'},limits:'Trusted hash-time identity, not execution authentication/hostile filesystem defense. Native fields quoted, no measurement/classifier/retry; exclusive retained-prefix output, not atomic/durable. PNG/visualQA require separate admission.'};
  writeFileSync(join(directory,'display-manifest.json'),JSON.stringify(manifest,null,2)+'\n',{flag:'wx'});
  // CLI table and persisted CSV consume exactly the same records, not a second extraction.
  return {manifest,cliTable:csv(records)};
}
