import test from 'node:test';
import assert from 'node:assert/strict';

// Break: JSON projection collapses NaN/null, -0/+0 or typed constructors.
test('exact comparison preserves nonfinite, negative zero and typed identity', () => {
  equalTree({a:NaN,b:-0,c:Uint8Array.of(0,255)}, {a:NaN,b:-0,c:Uint8Array.of(0,255)});
  for (const [a,b] of [[NaN,null],[-0,0],[Infinity,-Infinity],[Uint8Array.of(1),[1]]]) {
    assert.throws(() => equalTree(a,b), error => error.pointer === '');
  }
});
// Break: sorting/dropping unknown keys or reordering arrays hides differences.
test('exact comparison rejects key/order/extra-field changes at first pointer', () => {
  assert.throws(() => equalTree({a:1,b:2},{b:2,a:1}), /key order/);
  assert.throws(() => equalTree({a:[1,2]},{a:[2,1]}), e => e.pointer === '/a/0');
  assert.throws(() => equalTree({a:1},{a:1,elapsed:4}), /key order/);
});
// Break: time-name regex or subtree deletion exempts arbitrary semantic fields.
test('only explicitly supplied numeric timing leaves can differ', () => {
  const timingLeaves=[];
  equalTree({elapsed:1,value:2},{elapsed:3,value:2},'',new Set(['/elapsed']),timingLeaves);
  assert.deepEqual(timingLeaves,[{pointer:'/elapsed',left:1,right:3}]);
  assert.throws(() => equalTree({elapsed:1},{elapsed:'3'},'',new Set(['/elapsed'])), /finite/);
  assert.throws(() => equalTree({value:{elapsedMs:1}},{value:{elapsedMs:2}}), e => e.pointer === '/value/elapsedMs');
});

function clockFixture() {
  // Independent arithmetic: source shape182, sequential [2i,2i+1], elapsed1.
  return {
    intervals:Array.from({length:182},(_,i) => ({start:i*2,end:i*2+1,elapsed:1})),
    calculationLog:Array.from({length:176},(_,i) => ({nativeEntry:{elapsedMs:i%11===0?0.001:0.002}}))
  };
}
// Break: whitelisted numeric leaves skip finite/arithmetic/ordering checks.
test('clock validation rejects nonfinite, negative, inconsistent or reversed time', () => {
  validateTiming(clockFixture());
  for (const change of [s=>s.intervals[0].start=NaN,s=>s.intervals[0].elapsed=-0,
    s=>s.intervals[0].elapsed=2,s=>s.intervals[1].start=0,
    s=>s.calculationLog[0].nativeEntry.elapsedMs=0.0015,
    s=>s.calculationLog[0].nativeEntry.elapsedMs=0.003]) {
    const s=clockFixture(); change(s); assert.throws(() => validateTiming(s));
  }
});

// Break: digest declaration is trusted without binding actual bytes and length.
test('raw binding checks both actual digest and actual byte length', () => {
  const bytes=Uint8Array.of(1,2,3),sha256=createHash('sha256').update(bytes).digest('hex');
  verifyBinding(bytes,{bytes:3,sha256},'fixture');
  assert.throws(() => verifyBinding(bytes,{bytes:4,sha256},'fixture'));
  assert.throws(() => verifyBinding(bytes,{bytes:3,sha256:'0'.repeat(64)},'fixture'));
});
