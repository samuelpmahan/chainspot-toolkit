function readBound(path,expected,size,records,cap=33554432) {
  const fd=openSync(path,constants.O_RDONLY|constants.O_NONBLOCK);
  let raw;
  try {
    const s=fstatSync(fd);assert(s.isFile()&&Number.isSafeInteger(s.size)&&s.size>=0&&s.size<=cap,'regular bounded file required: '+path);
    if(size!==null)assert.equal(s.size,size,'byte count: '+path);
    raw=new Uint8Array(s.size);let offset=0;
    while(offset<raw.length){const n=readSync(fd,raw,offset,raw.length-offset,offset);assert(n>0,'short file: '+path);offset+=n;}
    assert.equal(readSync(fd,new Uint8Array(1),0,1,offset),0,'grown file: '+path);assert.equal(fstatSync(fd).size,s.size,'changed length: '+path);
  } finally {closeSync(fd);}
  const binding={path,bytes:raw.length,sha256:sha(raw)};
  if(expected!==null)assert.equal(binding.sha256,expected,'SHA256: '+path);
  records.push(binding);return raw;
}

// V8 supports undefined/NaN/-0/Uint8Array, but does not promise null-prototype
// object preservation. Carry an explicit prototype side table and verify a
// reconstructed round trip before calling this report format lossless.
function packV8(report,reviewed) {
  const nullPrototypePaths=[];
  const visit=(v,path)=>{
    if(v===null||typeof v!=='object'||v instanceof Uint8Array)return;
    if(Object.getPrototypeOf(v)===null)nullPrototypePaths.push(path);
    for(const key of Object.keys(v))visit(v[key],[...path,key]);
  };
  visit(report,[]);
  const envelope={format:'E1525-V8-REPORT+NULLPROTO@1',nullPrototypePaths,report};
  const bytes=serialize(envelope),copy=deserialize(bytes);
  for(const path of copy.nullPrototypePaths){let v=copy.report;for(const key of path)v=v[key];Object.setPrototypeOf(v,null);}
  if(reviewed)reviewed.equalTree(report,copy.report);else assert.deepStrictEqual(copy.report,report);
  return new Uint8Array(bytes);
}
