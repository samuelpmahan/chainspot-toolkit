export function equalTree(a,b,p='',allowed=new Set(),used=[]) {
  if(allowed.has(p)) { finite(a,p); finite(b,p); used.push({pointer:p,left:a,right:b}); return; }
  if(typeof a!=='object'||a===null||typeof b!=='object'||b===null) {
    requireThat(Object.is(a,b),'exact value differs',p,a,b); return;
  }
  const ap=Object.getPrototypeOf(a),bp=Object.getPrototypeOf(b);
  requireThat(ap===bp,'object/typed constructor differs',p,a,b);
  requireThat(ap===Object.prototype||ap===null||ap===Array.prototype||ap===Uint8Array.prototype,'unsupported object prototype',p,a,b);
  const ak=Reflect.ownKeys(a),bk=Reflect.ownKeys(b);
  requireThat(ak.length===bk.length&&ak.every((k,i)=>typeof k==='string'&&k===bk[i]),'key order/length differs',p,ak,bk);
  if(ap===Array.prototype)requireThat(ak.length===a.length+1&&bk.length===b.length+1&&Array.from({length:a.length},(_,i)=>String(i)).every((k,i)=>ak[i]===k),'dense undecorated array required',p);
  if(ap===Uint8Array.prototype)requireThat(ak.length===a.length&&bk.length===b.length&&ak.every((k,i)=>k===String(i)),'undecorated Uint8Array required',p);
  for(const k of ak) {
    const ad=Object.getOwnPropertyDescriptor(a,k),bd=Object.getOwnPropertyDescriptor(b,k);
    requireThat(Object.hasOwn(ad,'value')&&Object.hasOwn(bd,'value')&&ad.enumerable===bd.enumerable,'data property required',ptr(p,k));
    equalTree(ad.value,bd.value,ptr(p,k),allowed,used);
  }
}
export function verifyBinding(bytes,binding,p='') {

  const actual=sha(bytes); requireThat(actual===binding.sha256,'SHA256 binding differs',p,actual,binding.sha256);
  return {bytes:bytes.length,sha256:actual};
}

function inspectDirectory(root) {
  requireThat(isAbsolute(root)&&resolve(root)===root&&!lstatSync(root).isSymbolicLink()&&lstatSync(root).isDirectory(),'exact absolute nonsymlink output root required',root);
  const found=[];
  for(const name of readdirSync(root)) {
    const path=join(root,name),s=lstatSync(path);requireThat(!s.isSymbolicLink(),'output symlink refused',path);
    if(name==='fragments'&&s.isDirectory()) for(const child of readdirSync(path)) {requireThat(lstatSync(join(path,child)).isFile()&&!lstatSync(join(path,child)).isSymbolicLink(),'regular fragment required',join(path,child));found.push('fragments/'+child);}
    else {requireThat(s.isFile(),'unexpected output directory/type',path);found.push(name);}
  }
  const allowed=new Set([...artifactModel().map(x=>x.path),'artifact-manifest.json','cli-receipt.json']);
  requireThat(found.length===allowed.size&&found.every(p=>allowed.has(p)),'extra/missing output artifacts',root,found,[...allowed]);
}

/** Only entry point that reads files. Never writes, retries, measures or imports.
 * Admission bytes and outer records must be preserved by the root's independent
 * packet; this comparator checks their contents, not the honesty of a caller.
 * Actual chronology/fresh directory reservation remain outer-packet authority.
 */
export function comparePackages({leftRoot,rightRoot,policyPath,admission:a,codec}) {
  const report={schema:'E1519-pure-comparison@1',status:'NOT_COMPARABLE',policySha256:POLICY_SHA256,checks:[],hashes:[],timingLeaves:[],rawTiming:[],firstDifference:null,
    externalAuthority:'outer packet must independently attest observations, exact runtime, original-before-replay chronology and fresh external output roots; hashes do not authenticate executed code',
    claims:{scientificAcceptance:'UNKNOWN',humanAcceptance:'UNKNOWN',pathIdentity:'UNKNOWN',continuity:'UNKNOWN',calibratedWidth:'UNKNOWN'}};
  let stage='admission';
  const checked=name=>report.checks.push({check:name,status:'PASS'});
  try {
    const policy=readBytes(policyPath,32768);verifyBinding(policy,{bytes:14912,sha256:POLICY_SHA256},policyPath);report.hashes.push({path:policyPath,bytes:policy.length,sha256:sha(policy)});
    // E1512 wording defect is NOT a silently reinterpreted policy PASS.
    // Root must supply its append-only, reviewed clarification and its separately
    // admitted digest. This original-policy hash never changes.
    requireThat(a?.sourceShapeAddendum&&a.sourceShapeAddendum.admitted===true,'root-reviewed source-shape addendum required; original E1512 defect retained');
    const addendum=a.sourceShapeAddendum;
    requireThat(typeof addendum.sha256==='string'&&/^[0-9a-f]{64}$/.test(addendum.sha256),'admitted addendum digest required');
    const addendumBytes=readBytes(addendum.path,32768);verifyBinding(addendumBytes,{bytes:addendum.bytes,sha256:addendum.sha256},addendum.path);
    report.hashes.push({path:addendum.path,bytes:addendumBytes.length,sha256:sha(addendumBytes),authority:'root-reviewed source-shape clarification; original policy defect retained'});
    requireThat(codec&&typeof codec.decodeBytes==='function'&&typeof codec.encodeBytes==='function','separately admitted unchanged codec namespace required');
    requireThat(a&&typeof a.nodeExecutable==='string'&&isAbsolute(a.nodeExecutable),'admitted node executable required');
    keys(a.runtime,['node','platform','arch']);for(const k of ['node','platform','arch'])requireThat(typeof a.runtime[k]==='string'&&a.runtime[k].length>0,'runtime identity required');
    requireThat(leftRoot!==rightRoot,'distinct original/replay output roots required');
    for(const r of [leftRoot,rightRoot])requireThat(isExternalRoot(a.sourceRoot,r),'external output root required',r);
    const expected={requestBinding:requestFromAdmission(a,report.hashes),sourceBinding:checkSource(a.sourceRoot,report.hashes),runtime:a.runtime};checked(stage);
    stage='left independently valid';const left=readPackage(leftRoot,codec,expected,report.hashes);validateOuter(a.outer[0],left,a,report.hashes);checked(stage);
    report.rawTiming.push({root:leftRoot,intervals:left.sidecar.intervals,calculationLog:left.sidecar.calculationLog});
    stage='right independently valid';const right=readPackage(rightRoot,codec,expected,report.hashes);validateOuter(a.outer[1],right,a,report.hashes);checked(stage);
    report.rawTiming.push({root:rightRoot,intervals:right.sidecar.intervals,calculationLog:right.sidecar.calculationLog});
    stage='value and answer exact bytes/trees';
    for(const path of ['value-trace.json',...Array.from({length:160},(_,i)=>`fragments/answer-${i}.json`)]) {equalBytes(left.raw.get(path),right.raw.get(path),path);equalTree(left.decoded.get(path),right.decoded.get(path),path);}
    checked(stage);stage='named timing leaves only';
    const sideUsed=[];equalTree(left.sidecar,right.sidecar,'',timingPaths(),sideUsed);report.timingLeaves.push(...sideUsed.map(x=>({file:'timing-sidecar.json',...x})));
    for(let j=0;j<176;j++) {const path=`fragments/calculation-${j}.json`,used=[];equalTree(left.decoded.get(path),right.decoded.get(path),'',new Set(['/nativeEntry/elapsedMs']),used);report.timingLeaves.push(...used.map(x=>({file:path,...x})));}

    // and actual raw file bindings above. Only these now-proven dependencies
    // can differ, not any other metadata or a hash of an unchecked file.
    const allowedManifest=new Set(),allowedReceipt=new Set(['/artifactManifest/bytes','/artifactManifest/sha256']);
    artifactModel().forEach((m,i)=>{if(m.kind==='calculation'||m.kind==='timing-sidecar')for(const key of ['bytes','sha256']) {allowedManifest.add(`/artifacts/${i}/${key}`);allowedReceipt.add(`/artifacts/${i}/${key}`);}});
