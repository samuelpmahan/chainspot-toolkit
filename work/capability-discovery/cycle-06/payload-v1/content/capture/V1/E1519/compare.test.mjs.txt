test('exact comparison preserves nonfinite, negative zero and typed identity', () => {
  equalTree({a:NaN,b:-0,c:Uint8Array.of(0,255)}, {a:NaN,b:-0,c:Uint8Array.of(0,255)});
  for (const [a,b] of [[NaN,null],[-0,0],[Infinity,-Infinity],[Uint8Array.of(1),[1]]]) {
    assert.throws(() => equalTree(a,b), error => error.pointer === '');
  }

// Break: sorting/dropping unknown keys or reordering arrays hides differences.
test('exact comparison rejects key/order/extra-field changes at first pointer', () => {
  assert.throws(() => equalTree({a:1,b:2},{b:2,a:1}), /key order/);
  assert.throws(() => equalTree({a:[1,2]},{a:[2,1]}), e => e.pointer === '/a/0');
  assert.throws(() => equalTree({a:1},{a:1,elapsed:4}), /key order/);
});
// Break: time-name regex or subtree deletion exempts arbitrary semantic fields.
test('only explicitly supplied numeric timing leaves can differ', () => {
  const timingLeaves=[];
  equalTree({elapsed:1,value:2},{elapsed:3,value:2},'',new Set(['/elapsed']),timingLeaves);
  assert.deepEqual(timingLeaves,[{pointer:'/elapsed',left:1,right:3}]);
  assert.throws(() => equalTree({elapsed:1},{elapsed:'3'},'',new Set(['/elapsed'])), /finite/);
  assert.throws(() => equalTree({value:{elapsedMs:1}},{value:{elapsedMs:2}}), e => e.pointer === '/value/elapsedMs');
