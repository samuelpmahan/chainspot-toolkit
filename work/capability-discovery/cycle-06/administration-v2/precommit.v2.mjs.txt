const git=(...a)=>execFileSync('git',a,{cwd:REPO,maxBuffer:64*1024*1024});
const str=(...a)=>git(...a).toString('utf8').trim();
const nul=b=>{const s=b.toString('utf8');assert.ok(Buffer.from(s).equals(b));assert.ok(!s||s.endsWith('\0'));return s?s.slice(0,-1).split('\0'):[];};
const safe=p=>typeof p==='string'&&p&&!p.startsWith('/')&&!/[\\\x00-\x1f\x7f]/.test(p)&&p.split('/').every(x=>x&&x!=='.'&&x!=='..');
const bytes=p=>{const f=join(REPO,p),s=lstatSync(f);assert.ok(s.isFile()&&s.nlink===1&&realpathSync(f)===f,p);assert.equal(s.mode&0o7777,0o644,p);return readFileSync(f);};

assert.equal(str('symbolic-ref','--quiet','--short','HEAD'),'codex/capability-discovery-lab');

for(const e of entries){const b=bytes(e.path);assert.equal(b.length,e.bytes,e.path);assert.equal(sha(b),e.sha256,e.path);if(mode==='staged')assert.deepEqual(index.get(e.path),{mode:'100644',blob:blob(b),stage:'0'},e.path);}
