const sha=b=>createHash('sha256').update(b).digest('hex');
const blobSha=b=>createHash('sha1').update(`blob ${b.length}\0`).update(b).digest('hex');
const asciiJson=v=>JSON.stringify(v,null,2).replace(/[^\x00-\x7f]/g,c=>'\\u'+c.charCodeAt(0).toString(16).padStart(4,'0'))+'\n';
const git=(...a)=>execFileSync('git',a,{cwd:REPO,maxBuffer:64*1024*1024});

  const [mode,type,oid]=string('ls-tree',commit,'--',path).split(/\s+/);assert.equal(type,'blob');assert.equal(mode,'100644');
  const bytes=git('cat-file','blob',oid);assert.equal(blobSha(bytes),oid);
  const identity={path,mode,gitBlobSha1:oid,sha256:sha(bytes),byteLength:bytes.length};
  let text=null;try{const decoded=new TextDecoder('utf-8',{fatal:true,ignoreBOM:true}).decode(bytes);if(Buffer.from(decoded).equals(bytes))text=Array.from(decoded);}catch{}
  sequence++;const chunks=[];const rebuilt=[];
  const count=Math.max(1,Math.ceil((text===null?bytes.length:text.length)/(text===null?36000:4000)));
  for(let i=0;i<count;i++){
    const name=`${text===null?'binary':'text'}-${String(sequence).padStart(4,'0')}-${String(i+1).padStart(4,'0')}.json`;
    if(text===null){const chunk=bytes.subarray(i*36000,Math.min(bytes.length,(i+1)*36000));await put(name,{kind:'binary-base64-segment',...identity,segmentIndex:i+1,segmentCount:count,chunkByteLength:chunk.length,chunkSha256:sha(chunk),base64:chunk.toString('base64')});}

    const piece=JSON.parse(await readFile(join(out,name)));rebuilt.push(text===null?Buffer.from(piece.base64,'base64'):Buffer.from(piece.content));
  }
  const joined=Buffer.concat(rebuilt);assert.ok(joined.equals(bytes));assert.equal(blobSha(joined),oid);
  files.push({...identity,encoding:text===null?'binary-base64-segments':'utf8-text-segments',chunks});
}
