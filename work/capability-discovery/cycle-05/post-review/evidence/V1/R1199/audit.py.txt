def check(name,value):
    if not value: raise AssertionError(name)

def put(path,data):
    path.parent.mkdir(parents=True,exist_ok=True)

def dump(path,obj):put(path,(json.dumps(obj,indent=2)+'\n').encode())
def command(name,argv,cwd=REPO,expected=0):
    done=subprocess.run(argv,cwd=cwd,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    put(ROOT/(name+'.stdout.log'),done.stdout);put(ROOT/(name+'.stderr.log'),done.stderr)
    entry={'name':name,'argv':list(map(str,argv)),'cwd':str(cwd),'exitCode':done.returncode,'expected':expected,'stdoutSha256':sha(done.stdout),'stderrSha256':sha(done.stderr)}
    report['commands'].append(entry)
    if expected is not None:check(name+' exit code',done.returncode==expected)
    return done
