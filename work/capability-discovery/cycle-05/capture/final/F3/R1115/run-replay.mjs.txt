// Exactly one independent CLI replay. No parameter/pixel/source changes.
import {readFileSync,writeFileSync,existsSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {spawnSync} from 'node:child_process';
import {fileURLToPath} from 'node:url';
import {performance} from 'node:perf_hooks';
import assert from 'node:assert/strict';
const root=new URL('../../../',import.meta.url),here=new URL('./',import.meta.url);
const sha=b=>createHash('sha256').update(b).digest('hex'),read=p=>readFileSync(new URL(p,root));
const fixed={
 'discovery-cycle05/F5/E1115/result.json':'edff8f00944ad3fff90d92d1cbd68d5802bba7da39844671e2907c43083c015c',
 'discovery-cycle05/F5/E1115/run/value-trace.json':'76cf61ee3be3cd56faa0fc7274d5a3a5b898fe954dc73571092549487fa11eb2',
 'discovery-cycle05/F5/E1115/run/timing-sidecar.json':'4a6599406ed17ccd97a3c8e377d0186a0fb8f6fcc064189439870b83b70f24f9',
 'discovery-cycle05/F6/E1111/source-freeze.json':'0a2351f595fd72d55af79fccb4532deb1b58125a2c440e37d6276fba5d21da4b',
 'discovery-cycle05/numeric-comparison-freeze.json':'8642e4adfc1623eb4de0ab0a1af4ac75053239cfa623c67ede3cd5f5c827bf3c',
 'discovery-cycle05/F5/E1115/manifest.json':'3a27691d56bd32cb22c5f522b4114d3eaadc5adebc2a57740439eceff5200ed2'};
const freeze=JSON.parse(read('discovery-cycle05/F6/E1111/source-freeze.json'));
for(const f of freeze.files)fixed['chainspot-discovery-lab/'+f.path]=f.sha256;
const contract=JSON.parse(read('discovery-cycle05/F5/E1110/executable-contract.json'));
for(const f of contract.route.modules)fixed['chainspot-discovery-lab/'+contract.route.moduleBase+f.path]=f.sha256;
const inventory=JSON.parse(read('discovery-cycle05/F4/E1102/inventory.json'));
fixed['discovery-cycle05/F4/E1102/protocol.json']=inventory.protocolFileSha256;
fixed['discovery-cycle05/F4/E1102/inventory.json']='c103c508e5cb8668bbc5069620046c3f5b2c3eea05d018af9d17169e308529aa';
for(const f of Object.values(inventory.rasters))fixed['discovery-cycle05/F4/E1102/'+f.rawPath]=f.rgbaSha256;
for(const f of Object.values(inventory.masks))fixed['discovery-cycle05/F4/E1102/'+f.path]=f.sha256;
const check=()=>Object.entries(fixed).map(([path,expectedSha256])=>{const bytes=read(path),actualSha256=sha(bytes);assert.equal(actualSha256,expectedSha256,path);return{path,bytes:bytes.length,expectedSha256,actualSha256};});
const before=check();assert(!existsSync(new URL('replay-run',here)),'exclusive replay output already exists');
const command={executable:process.execPath,cwd:fileURLToPath(new URL('chainspot-discovery-lab/',root)),args:[
 'experiments/capability-discovery/station-evidence/cli.mjs','run','--manifest',fileURLToPath(new URL('discovery-cycle05/F5/E1115/manifest.json',root)),
 '--out',fileURLToPath(new URL('replay-run',here))]};
writeFileSync(new URL('replay-command.json',here),JSON.stringify(command,null,2)+'\n',{flag:'wx'});
const startUTC=new Date().toISOString(),start=performance.now();
writeFileSync(new URL('replay-start.json',here),JSON.stringify({startUTC,command},null,2)+'\n',{flag:'wx'});
const child=spawnSync(command.executable,command.args,{cwd:command.cwd,encoding:'utf8',maxBuffer:1048576});
const elapsedMs=performance.now()-start,endUTC=new Date().toISOString();
writeFileSync(new URL('replay.stdout.log',here),child.stdout??'',{flag:'wx'});
writeFileSync(new URL('replay.stderr.log',here),child.stderr??'',{flag:'wx'});
const after=check();
const receipt={schema:'R1115-one-independent-CLI-replay@1',job_id:'R1115',worker:'F3',runs:1,startUTC,endUTC,elapsedMs,
 exitCode:child.status,signal:child.signal,error:child.error?{name:child.error.name,message:child.error.message}:null,command,before,after};
writeFileSync(new URL('replay-process.json',here),JSON.stringify(receipt,null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify({runs:1,startUTC,endUTC,elapsedMs,exitCode:child.status,stdout:child.stdout,stderr:child.stderr,checkedBindings:before.length}));
process.exitCode=child.status??1;
