const sha=b=>createHash('sha256').update(b).digest('hex');
export const blobSha=b=>createHash('sha1').update(Buffer.from(`blob ${b.length}\0`)).update(b).digest('hex');
export const asciiJson=v=>JSON.stringify(v).replace(/[\u007f-\uffff]/g,c=>'\\u'+c.charCodeAt(0).toString(16).padStart(4,'0'))+'\n';
export function textChunks(value,budget=60000){
  if(budget<12)throw Error('Chunk budget too small');
  const chunks=[];let s='',cost=0;
  for(const cp of value){const n=asciiJson(cp).length-3;if(cost+n>budget){chunks.push(s);s='';cost=0;}s+=cp;cost+=n;}
  if(s||!chunks.length)chunks.push(s);
  return chunks;
}
export function pages(entries,budget=90000){
  const result=[];let p=[];
  for(const entry of entries){if(asciiJson([entry]).length>budget)throw Error('Oversized index entry');if(asciiJson([...p,entry]).length>budget){result.push(p);p=[];}p.push(entry);}
  if(p.length)result.push(p);return result;
}
const allowed=p=>p.startsWith(CYCLE)||(p.startsWith(HELPER)&&HELPER_FILES.has(p.slice(HELPER.length)));

function selfTest(){
  assert.equal(blobSha(Buffer.alloc(0)),'e69de29bb2d1d6434b8b29ae775ad8c2e48c5391');
  assert.equal(blobSha(Buffer.from('hello\n')),'ce013625030ba8dba906f756967f9e9ca394464a');
  for(const original of ['', 'ascii\n"\\', 'a😀é\u007f'.repeat(22000)]){
    const parts=textChunks(original);assert.equal(parts.join(''),original);
    for(const p of parts){assert.ok(asciiJson({content:p}).length<90000);assert.match(asciiJson(p),/^[\x00-\x7f]*$/);assert.equal(JSON.parse(asciiJson(p)),p);assert.ok(!/[\ud800-\udbff]$/.test(p));assert.ok(!/^[\udc00-\udfff]/.test(p));}
  }
  const entries=Array.from({length:50},(_,i)=>({path:'file'+i,content:'x'.repeat(3000)}));
  const grouped=pages(entries,10000);assert.deepEqual(grouped.flat(),entries);assert.ok(grouped.every(p=>asciiJson(p).length<=10000));
  assert.throws(()=>pages([{x:'a'.repeat(100)}],20));
  assert.ok(allowed(CYCLE+'POST-REVIEW.md'));assert.ok(allowed(HELPER+'group.mjs'));assert.ok(!allowed(HELPER+'surprise.mjs'));assert.ok(!allowed('README.md'));
  console.log('transport self-test: 2 Git fixtures, Unicode/ASCII/budget/ordering/scope checks PASS; no repository writes');
}
