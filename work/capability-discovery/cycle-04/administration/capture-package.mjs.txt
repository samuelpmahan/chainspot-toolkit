/** Mechanical archival copy only; no CV imports/calls or original-file mutation. */
import {readdirSync,readFileSync,writeFileSync,mkdirSync,existsSync} from 'node:fs';
import {resolve,join,relative,dirname,basename} from 'node:path';
import {createHash} from 'node:crypto';
const from=resolve(process.argv[2]),to=resolve(process.argv[3]);
if(!from.endsWith('/discovery-cycle04')||!to.endsWith('/work/capability-discovery/cycle-04')||existsSync(to))throw Error('Expected exact source and fresh cycle-04 package');
const sha=b=>createHash('sha256').update(b).digest('hex');
const put=(p,b)=>{mkdirSync(dirname(p),{recursive:true});writeFileSync(p,b,{flag:'wx'});};
const all=[];
function walk(d){for(const e of readdirSync(d,{withFileTypes:true})){const p=join(d,e.name);if(e.isDirectory())walk(p);else if(e.isFile())all.push(relative(from,p));}}
walk(from);mkdirSync(to,{recursive:true});
const files=[],omitted=[],unique=new Map(),templates=new Map();
const priority=p=>/^P(?:2\/E811|3\/E809|4\/E807|5\/E810)\//.test(p)?0:1;
all.sort((a,b)=>priority(a)-priority(b)||a.localeCompare(b));
for(const original of all){
  if(original.startsWith('P6/E899/'))continue; // This active administrative job is recorded separately, never recursively copied.
  const source=join(from,original),bytes=readFileSync(source),record={original,sha256:sha(bytes),bytes:bytes.length};
  if(original.includes('__pycache__/')||original.includes('stager-generated-pycache/')||original.endsWith('.pyc')){omitted.push({...record,reason:'generated bytecode cache; not original source or measurement evidence'});continue;}
  const bulkJson=/^request(?:-[AB])?\.json$/.test(basename(original))||basename(original)==='source.json'||basename(original)==='staged-source.json'||original.endsWith('-source.json');
  if(bulkJson){
    let descriptor=templates.get(record.sha256);
    if(!descriptor){
      const text=bytes.toString('utf8'),value=JSON.parse(text),s=value.source??value,image=s.image,data=image?.data;
      if(!Array.isArray(data)||!data.every(x=>Number.isInteger(x)&&x>=0&&x<=255))throw Error('Unrecognized bulk data: '+original);
      const m=/"data"\s*:\s*\[/.exec(text);if(!m)throw Error('No raw pixel array: '+original);
      const start=m.index+m[0].length-1,end=text.indexOf(']',start),raw=text.slice(start,end+1);
      const arrayPrefix=/^\[\s*/.exec(raw)[0],separator=/,\s*/.exec(raw)?.[0]??',',arraySuffix=/\s*\]$/.exec(raw)[0];
      if(arrayPrefix+data.join(separator)+arraySuffix!==raw)throw Error('Nonuniform raw array serialization: '+original);
      descriptor='bulk-descriptors/'+record.sha256+'.json';
      const desc={schema:'cycle04-omitted-pixel-array@1',originalSha256:record.sha256,originalBytes:bytes.length,prefix:text.slice(0,start),suffix:text.slice(end+1),arrayPrefix,separator,arraySuffix,length:data.length,rgbaSha256:sha(Uint8Array.from(data)),image:{width:image.width,height:image.height,frame:image.frame},provenance:s.provenance??value.provenance};
      put(join(to,descriptor),JSON.stringify(desc,null,2)+'\n');templates.set(record.sha256,descriptor);
    }
    omitted.push({...record,reason:'bulk pixel JSON omitted; exact nonpixel prefix/suffix and array serialization retained for byte-exact reconstruction without CV',descriptor});continue;
  }
  if(original.endsWith('.png')&&bytes.length>2_000_000){omitted.push({...record,reason:'external original source PNG; exact platform Git blob identity in source-manifest.json'});continue;}
  const previous=unique.get(record.sha256);
  if(previous){files.push({...record,path:previous,storage:'byte-identical alias; restored at original path by stage-replay.py'});continue;}
  const stored='capture/'+original+(original.endsWith('.mjs')?'.txt':'');
  put(join(to,stored),bytes);unique.set(record.sha256,stored);
  files.push({...record,path:stored,storage:original.endsWith('.mjs')?'exact original bytes archived as .mjs.txt; original filename restored only in scratch':'exact original bytes'});
}
const quarantine=join(from,'P6/E899/quarantine/truncated-R812-counterexample.json.partial');
if(existsSync(quarantine))put(join(to,'administration/rejected-R812-copy.json.partial.txt'),readFileSync(quarantine));
put(join(to,'administration/capture-index.json'),JSON.stringify({schema:'cycle04-capture-index@1',sourceRoot:from,files,omitted,scope:'All captured originals are byte-exact. Archival suffix and deduplication are filename/storage mappings only. Active P6/E899 administrative work is recorded separately; no HH source captured.'},null,2)+'\n');
console.log(JSON.stringify({package:to,capturedOriginals:files.length,uniqueStoredFiles:unique.size,omitted:omitted.length,bulkDescriptors:templates.size,packageFreeze:false}));
