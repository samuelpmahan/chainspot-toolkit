/** R807 independent replay/audit. Only writes below this review's own directory. */
import { readFileSync, writeFileSync, mkdirSync, existsSync } from 'node:fs';
import { createHash } from 'node:crypto';

const json=p=>JSON.parse(read(p));
const checks=[];

const put=(name,value)=>{const b=Buffer.from(JSON.stringify(value,null,2)+'\n');writeFileSync(join(out,name),b,{flag:'wx'});return {path:join(out,name),sha256:sha(b),bytes:b.length};};

const baselineFiles={};

const startedAt=new Date().toISOString();
try {
  if(existsSync(join(out,'verification.json'))||existsSync(join(out,'failure.json'))||existsSync(replay))throw Error('Review replay already exists; no retry');
  const resultBytes=read(join(original,'result.json')),runBytes=read(join(original,'run.json'));
  check('exact assigned producer result',sha(resultBytes)==='63955be1fd02b4a2e2bc1ec1bd1ed8ec1803e476b85e3e3d77179a71ae91c84a',sha(resultBytes));
  check('exact assigned producer run',sha(runBytes)==='63c5e408df83fe53e0ac993c297aad5a1a41410af0dc4ca5e609779f36c249db',sha(runBytes));

  check('reviewed producer source',sha(producerBytes)===checkpoint['producer.mjs'],sha(producerBytes));
  check('reviewed renderer source',sha(rendererBytes)===checkpoint['render-receipt.py'],sha(rendererBytes));
  for(const a of result.artifacts){checked(original,a);baselineFiles[join(original,a.path)]=a.sha256;}
  baselineFiles[join(original,'result.json')]=sha(resultBytes);
  for(const [p,h] of Object.entries(run.sourceFreeze)){check('upstream source freeze '+p,sha(read(p))===h,h);baselineFiles[p]=h;}
  for(const [p,h] of Object.entries(freeze.sourceHashes)){check('frozen source '+p,sha(read(join(repo,p)))===h,h);baselineFiles[join(repo,p)]=h;}
  const d31path='work/capability-discovery/cycle-03/evidence/E506/candidate-freeze.json';
  const acceptedPath='work/capability-discovery/cycle-02/feature-manifest-E308.json';
  const d31=json(join(repo,d31path)),accepted=json(join(repo,acceptedPath));
  check('accepted source-pixel freeze',sha(read(join(repo,d31path)))==='d31b7a4312ac1bfc4f76bc96fc278f8acad4f49f4b04283df0faa8e6dce54626',d31path);
  check('accepted composition freeze',sha(read(join(repo,acceptedPath)))==='5b59dce1c7ae87f962c0305697ed883e423bd504fedce5851fb02242309dcf52',acceptedPath);
  const closure=[...d31.files,...d31.retainedContractHistoryAndEvidence,...accepted.files,accepted.existingGuard,...accepted.immutableSources.map(s=>({path:'experiments/capability-discovery/probes/width/'+s.file,sha256:s.sha256}))];
  for(const a of closure){check('accepted closure '+a.path,sha(read(join(repo,a.path)))===a.sha256,a.sha256);baselineFiles[join(repo,a.path)]=a.sha256;}
  for(const c of run.cases)for(const a of[c.request,c.receipt,c.binding]){checked(original,a);baselineFiles[join(original,a.path)]=a.sha256;}
  const originalComparison=JSON.parse(checked(original,run.comparison));
  check('94 original literal checks',originalComparison.checks.length===94&&originalComparison.checks.every(c=>c.pass&&eq(c.actual,c.expected)),{checks:originalComparison.checks.length,failed:originalComparison.failedChecks});

  let portable=producerBytes.toString('utf8');
  const changes=[
    ["new URL('../../P5/fixtures.mjs',import.meta.url)","new URL('file:///workspace/scratch/8528bb251845/discovery-cycle04/P5/fixtures.mjs')"],
    ["new URL('../../../chainspot-discovery-lab/experiments/capability-discovery/composition/recipe.mjs',import.meta.url)","new URL('file:///workspace/scratch/8528bb251845/chainspot-discovery-lab/experiments/capability-discovery/composition/recipe.mjs')"]

Output remains script-local under reviewer replay. Scientific code, fixture bytes, source settings, comparisons and renderer remain unchanged.',metadata:'Replay files intentionally retain original E807/P4 schema fields for byte-equivalent reproduction; this record binds their separate R807 reviewer provenance.'});
  const proc=spawnSync(process.execPath,[join(replay,'producer.mjs')],{cwd:replay,encoding:'utf8',maxBuffer:10000000});
  writeFileSync(join(out,'run.stdout.log'),proc.stdout??'',{flag:'wx'});writeFileSync(join(out,'run.stderr.log'),proc.stderr??'',{flag:'wx'});
  const replayRun=json(join(replay,'run.json'));compositionCalls=replayRun.compositionCalls;
  check('replay producer exits zero',proc.status===0,{exitCode:proc.status,error:proc.error?.message});
  check('exactly four fresh calls',compositionCalls===4&&eq(replayRun.caseOrder,run.caseOrder),{compositionCalls,caseOrder:replayRun.caseOrder});

    caseBindings.push({id,request:actual.request,receipt:actual.receipt,rgbaSha256:binding.rgbaSha256,scientificBindingMatched:true,replayProcessObservation:rp,originalProcessObservation:op});
